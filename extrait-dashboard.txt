function remplirSelect(
    id,
    nouvelleValeur
){

    if(!nouvelleValeur){
        return;
    }


    const select =
        document.getElementById(
            id
        );


    const recherche =
        String(
            nouvelleValeur
        )
        .trim()
        .toLowerCase();


    const option =
        Array.from(
            select.options
        )
        .find(
            option =>
                option.value
                    .toLowerCase()
                === recherche
        );


    if(option){

        select.value =
            option.value;
    }
}
